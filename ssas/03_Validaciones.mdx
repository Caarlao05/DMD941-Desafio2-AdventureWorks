-- Ejecutar cada SELECT por separado en SSMS conectado a Analysis Services.
-- Base SSAS: DMD941_Desafio2. Cubo: CuboVentasInternet.
-- total general.
SELECT {[Measures].[Ventas],[Measures].[Unidades],
 [Measures].[Pedidos distintos],[Measures].[Lineas],
 [Measures].[Ticket promedio]} ON COLUMNS
FROM [CuboVentasInternet];

-- ventas por anio.
SELECT {[Measures].[Ventas],[Measures].[Unidades]} ON COLUMNS,
 NON EMPTY [Tiempo].[Anio].[Anio].Members ON ROWS
FROM [CuboVentasInternet];

-- matriz anio x pais.
SELECT {[Measures].[Ventas]} ON COLUMNS,
 NON EMPTY CrossJoin([Tiempo].[Anio].[Anio].Members,
 [Territorio].[Pais].[Pais].Members) ON ROWS
FROM [CuboVentasInternet];

-- categorias.
SELECT {[Measures].[Ventas],[Measures].[Unidades]} ON COLUMNS,
 NON EMPTY [Producto].[Categoria].[Categoria].Members ON ROWS
FROM [CuboVentasInternet];

-- diez claves de producto con mas ventas.
-- En empates, el ultimo puesto puede diferir: SQL tiene desempate por nombre/clave.
SELECT {[Measures].[Ventas],[Measures].[Unidades]} ON COLUMNS,
 TopCount(NonEmpty([Producto].[Producto].[Producto].Members,
 [Measures].[Ventas]),10,[Measures].[Ventas]) ON ROWS
FROM [CuboVentasInternet];

-- evolucion mensual cronologica.
SELECT {[Measures].[Ventas]} ON COLUMNS,
 NON EMPTY [Tiempo].[Mes].[Mes].Members ON ROWS
FROM [CuboVentasInternet];
